1Go to supabase.com → Start your project (free, no credit card)
2Create a new project → wait ~1 min to spin up
3Go to SQL Editor → paste this entire block and click Run:
create table if not exists expenses (id text primary key, data jsonb);
create table if not exists members  (id text primary key, list jsonb);
create table if not exists flat_expenses (id text primary key, data jsonb);
alter table expenses       enable row level security;
alter table members        enable row level security;
alter table flat_expenses  enable row level security;
create policy "allow all" on expenses      for all using (true) with check (true);
create policy "allow all" on members       for all using (true) with check (true);
create policy "allow all" on flat_expenses for all using (true) with check (true);
